Zero-Dependency Magic Bytes Detection

Detects 30+ file signatures directly from byte slices without third-party dependencies:

package main

import (
    "fmt"
    "github.com/PabloCirre/kaijuconverter-go"
)

func main() {
    header := []byte("%PDF-1.7...")
    format, ok := kaijuconverter.DetectMagicBytes(header)
    if ok {
        fmt.Printf("Detected %s (%s) -> %s\n", format.Extension, format.MIMEType, format.DocURL)
    }
}
